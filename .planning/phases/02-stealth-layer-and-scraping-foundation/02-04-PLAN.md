  ```python
  """Telegram alert service for scraper failure notifications."""
  import os
  import logging
  from datetime import datetime, timezone

  logger = logging.getLogger(__name__)


  class AlertService:
      """Sends one-shot Telegram alerts on scraper failures.

      Uses python-telegram-bot Bot class with async context manager.
      Reads TELEGRAM_BOT_TOKEN and NEMO_AUTH_USER_ID from environment.
      Alert format: "{platform} scraper: 0 results - {error_type} - {ISO timestamp}"
      """

      async def send_alert(self, platform: str, error_type: str) -> None:
          """Send a Telegram alert. Swallows all exceptions."""
          token = os.getenv("TELEGRAM_BOT_TOKEN")
          chat_id = os.getenv("NEMO_AUTH_USER_ID")
          if not token or not chat_id:
              logger.warning("Telegram credentials missing (TELEGRAM_BOT_TOKEN or NEMO_AUTH_USER_ID) — cannot send alert")
              return

          text = (
              f"{platform} scraper: 0 results - {error_type} - "
              f"{datetime.now(timezone.utc).strftime('%Y-%m-%dT%H:%M:%SZ')}"
          )
          try:
              from telegram import Bot
              async with Bot(token=token) as bot:
                  await bot.send_message(chat_id=int(chat_id), text=text)
              logger.info(f"Telegram alert sent: {text}")
          except Exception as e:
              logger.error(f"Failed to send Telegram alert: {e}")
              # Do NOT re-raise — alerting failure must not crash the cycle
  ```

  Key implementation details:
  - Import `Bot` inside the method to avoid import errors when telegram is not installed in test environments
  - Use `NEMO_AUTH_USER_ID` (NOT `TELEGRAM_CHAT_ID`) per STATE.md Phase 1 decision
  - Cast `chat_id` to `int()` because Telegram Bot API expects numeric ID
  - Use `datetime.now(timezone.utc)` not deprecated `datetime.utcnow()`
  - Swallow ALL exceptions in outer try/except — alerting must never crash the cycle

  import pytest
  from unittest.mock import patch, AsyncMock, MagicMock
  from src.alerter import AlertService


  @pytest.mark.asyncio
  async def test_alert_message_format():
      """SCRAPE-05: send_alert calls Bot.send_message with platform + timestamp."""
      mock_bot_instance = AsyncMock()
      mock_bot_cls = MagicMock()
      mock_bot_cls.return_value.__aenter__ = AsyncMock(return_value=mock_bot_instance)
      mock_bot_cls.return_value.__aexit__ = AsyncMock(return_value=False)

      with patch.dict("os.environ", {"TELEGRAM_BOT_TOKEN": "tok123", "NEMO_AUTH_USER_ID": "999"}):

              alerter = AlertService()
              await alerter.send_alert("LinkedIn", "possible block")

      mock_bot_instance.send_message.assert_called_once()
      call_kwargs = mock_bot_instance.send_message.call_args
      text = call_kwargs.kwargs.get("text") or call_kwargs[1].get("text") or call_kwargs[0][1] if len(call_kwargs[0]) > 1 else ""
      # Handle both positional and keyword args
      if not text:
          for arg in call_kwargs[0]:
              if isinstance(arg, str) and "LinkedIn" in arg:
                  text = arg
                  break
      if not text:
          text = str(call_kwargs)
      assert "LinkedIn" in text
      assert "possible block" in text
