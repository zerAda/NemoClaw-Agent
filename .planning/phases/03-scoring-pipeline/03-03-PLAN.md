  ```python
  async def process_job(self, listing: JobListing):
      """Score and process a single job listing. Persists score record regardless of outcome."""
      try:
          job_url = listing.url or ""
          if not job_url:
              logger.warning(f"Skipping listing with no URL: {listing.title}")
              return

          if self.memory.is_already_processed(job_url):
              logger.info(f"Skipping already seen job: {listing.title}")
              return

          # --- SCORE-01/02/03: Score the listing ---
          score_record = await self.hunter.score_job(listing)

          # --- SCORE-04: Persist score record for every job regardless of outcome ---
          self.memory.add_application(job_url, {
              "title": listing.title,
              "company": listing.company,
              "source": listing.source,
              "score": score_record.score,
              "recommendation": score_record.recommendation,
              "reasoning": score_record.reasoning,
              "fast_failed": score_record.fast_failed,
              "threshold_met": score_record.threshold_met,
              "status": "SKIPPED" if score_record.recommendation == "SKIP" else "SCORED",
          })

          # --- SCORE-03: Do not queue for application if below threshold ---
          if score_record.recommendation == "SKIP":
              logger.info(
                  f"SKIP: {listing.title} "
                  f"(score={score_record.score:.2f}, fast_failed={score_record.fast_failed})"
              )
              return

          # --- Generate cover letter for qualifying jobs ---
          logger.info(f"MATCH: {listing.title} (score={score_record.score:.2f}) — generating letter")
          jd_text = listing.description or listing.title
          letter = await self.tailor.customize_letter(jd_text, score_record.model_dump())

          job_id_uuid = self.memory._generate_uuid(job_url)
          output_dir = os.path.join(self.brain_path, "applications", job_id_uuid)
          os.makedirs(output_dir, exist_ok=True)

          with open(os.path.join(output_dir, "cover_letter.txt"), "w", encoding="utf-8") as f:
              f.write(letter.body)

          self.memory.add_application(job_url, {
              "title": listing.title,
              "company": listing.company,
              "source": listing.source,
              "score": score_record.score,
              "recommendation": score_record.recommendation,
              "reasoning": score_record.reasoning,
              "status": "READY",
              "uuid": job_id_uuid,
          })
          logger.info(f"ARTIFACT READY: {listing.title} -> {output_dir}")

      except Exception as e:
          logger.error(f"CRITICAL ERROR processing '{listing.title}': {e}")
  ```
